{Parametric equations}
%{ and center of mass}

\medskip

\begin{goal} \rm
Apply calculus to parametric equations
% and center of mass.
\end{goal}
%\smallskip

\begin{enumerate}
\item Write a parametric equation for a line of slope $\frac{4}{3}$ going through point $(-1, 2)$. 
\vfill
\item Write a parametric equation for a circle of radius $10$ with center $(4, -3)$.
\vfill
\pagebreak
\item  Consider the parametric equation $c(t) = (x(t), y(t)) = (4t -1, 2t^3)$.
\begin{enumerate}
	
\item Find a formula for the slope of a tangent line to the graph (as a function of $t$). What
is the slope when $t = 5$?
\vfill
\item Find the area between $c(t)$ and the $x$-axis from $t = 0$ to $t = 3$.
\vfill
\item What is the speed of a particle traveling along $c(t)$ for any value of $t$? What is the speed at $t = 0$?
\vfill
\item Set up but do not evaluate an integral to find the arc length of $c(t)$ for $-5 \leq t \leq 5$.
\vfill
\item Find the surface area of the surface of revolution created by revolving $c(t)$ for $0 \leq t \leq 2$ about the $x$-axis.
\vfill
\end{enumerate}
\end{enumerate}
