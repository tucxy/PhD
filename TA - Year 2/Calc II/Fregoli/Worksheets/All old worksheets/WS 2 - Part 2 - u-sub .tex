Get back into the swing of calculus with some integrals!

\begin{enumerate}
\item Evaluate the integral by direct antidifferentiation: 
\begin{enumerate}
\item
$\displaystyle\int \left(2x + 3x^{4/5}\right) \ dx$
\vfill
\item
$\displaystyle\int 2e^x \ dx$
\vfill
 \item
$\displaystyle\int \displaystyle\frac{10}{x} \ dx$
\vfill
\item
$\displaystyle\int \sec^2(x) \ dx$
\vfill
\item
$\displaystyle\int \displaystyle\frac{4}{\sqrt{1-x^2}}  \ dx$ (Hint: What are the derivatives of the inverse trig functions?)
\vfill
\end{enumerate}

\newpage

\item Use substitution to evaluate the following integrals. Write down the substitution you use.
\begin{enumerate}
\item $\displaystyle\int 4\cos(7x) \ dx$

\vspace{1.75in}

\item $\displaystyle\int \displaystyle\frac{5}{x \ln x} \ dx$

\vspace{1.75in}

\item $\displaystyle\int_0^1 x^2 \sqrt{4x^3+5} \ dx$

\vspace{1.75in}

\item $\displaystyle\int \displaystyle\frac{\cos(3x)}{e^{\sin(3x)}}\ dx$
\vfill
\newpage
\item $\displaystyle \int_0^x u^3 \sqrt{1 + u^2} \ du$
\vfill
\vfill
\end{enumerate}
